        // Obtener user_id de URL params o generar uno temporal
        function getUserId() {
            const urlParams = new URLSearchParams(window.location.search);
            const userId = urlParams.get('user_id');

            if (userId) {
                return userId;
            }

            // Si estamos en Braze/Appboy, intentar obtener el ID
            if (typeof appboyBridge !== 'undefined' && appboyBridge.getUser) {
                try {
                    const brazeUser = appboyBridge.getUser();
                    if (brazeUser && brazeUser.getUserId) {
                        return brazeUser.getUserId();
                    }
                } catch (e) {
                    console.log('No se pudo obtener user_id de Braze');
                }
            }

            // Fallback: generar ID temporal y guardarlo en sessionStorage
            let tempUserId = sessionStorage.getItem('temp_user_id');
            if (!tempUserId) {
                tempUserId = 'temp_' + Date.now() + '_' + Math.random().toString(36).substr(2, 9);
                sessionStorage.setItem('temp_user_id', tempUserId);
            }
            return tempUserId;
        }

        function toggleTerms() {
            state.termsAccepted = !state.termsAccepted;

            // Log de Braze: checkbox de términos
            if (typeof appboyBridge !== 'undefined') {
                if (state.termsAccepted) {
                    appboyBridge.logClick('Términos Aceptados - Checkbox');
                } else {
                    appboyBridge.logClick('Términos Rechazados - Checkbox');
                }
            }

            renderApp();
        }

        function toggleTermsModal(show) {
            const modal = document.getElementById('terms-modal');
            if (show) {
                modal.classList.remove('hidden');

                // Log de Braze: abrir modal de términos
                if (typeof appboyBridge !== 'undefined') {
                    appboyBridge.logClick('Ver Términos y Condiciones');
                }
            } else {
                modal.classList.add('hidden');
            }
        }

        function acceptTerms() {
            state.termsAccepted = true;
            toggleTermsModal(false);

            // Log de Braze: aceptar términos desde modal
            if (typeof appboyBridge !== 'undefined') {
                appboyBridge.logClick('Términos Aceptados - Botón Modal');
            }

            renderApp();
        }

        function closeInApp(e) {
            if (e) e.stopPropagation();
            if (typeof appboyBridge !== 'undefined') {
                appboyBridge.logClick('Cerrar Publicidad');
                appboyBridge.closeMessage();
                appboyBridge.logClick('Cerrar Inapp');
            }
        }

        // Llamar al API para abrir pack
        async function openPackFromAPI() {
            const userId = getUserId();

            try {
                const response = await fetch(`${API_BASE_URL}/open-pack`, {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/json'
                    },
                    body: JSON.stringify({
                        album_id: ALBUM_ID,
                        user_id: userId
                    })
                });

                const data = await response.json();

                if (!response.ok || !data.success) {
                    throw new Error(data.message || 'Error al abrir el pack');
                }

                return data.data;
            } catch (error) {
                console.error('Error abriendo pack:', error);
                throw error;
            }
        }

        // Registrar ganador en el API
        async function registerWinner(packData) {
            try {
                const response = await fetch(`${API_BASE_URL}/register-winner`, {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/json'
                    },
                    body: JSON.stringify(packData)
                });

                const data = await response.json();

                if (!response.ok || !data.success) {
                    console.error('Error registrando ganador:', data.message);
                    // No lanzamos error para no interrumpir la experiencia del usuario
                }

                console.log('Ganador registrado exitosamente');
                return data;
            } catch (error) {
                console.error('Error registrando ganador:', error);
                // No lanzamos error para no interrumpir la experiencia del usuario
            }
        }

        async function startOpeningSequence(e) {
            if (e) e.stopPropagation();
            if (!state.termsAccepted || state.isLoading) return;

            console.log('🎮 Iniciando apertura de pack...');

            if (typeof appboyBridge !== 'undefined') {
                appboyBridge.logClick('Abrir Pack');
            }

            // Activar estado de carga
            state.isLoading = true;
            renderApp(); // Re-renderizar para mostrar el loader

            const packContainer = document.getElementById('pack-container');

            try {
                // Llamar al API para abrir el pack
                console.log('📡 Llamando al API...');
                const packData = await openPackFromAPI();
                console.log('✅ Pack recibido del API:', packData);

                // Animación de cierre del pack
                if (packContainer) {
                    packContainer.style.transform = 'scale(3) rotate(20deg)';
                    packContainer.style.opacity = '0';
                }

                setTimeout(() => {
                    state.unopenedPacks--;

                    // Convertir los stickers del API al formato del frontend
                    const pickedStickers = packData.pack_stickers.map(sticker => ({
                        id: sticker.sticker_id,
                        url: sticker.sticker_url,
                        prize: sticker.prize_points,
                        brand: sticker.brand,
                        isPrize: sticker.is_prize,
                        prizeValue: sticker.prize_points
                    }));

                    console.log('🎨 Stickers procesados:', pickedStickers);

                    // Log de Braze: pack abierto exitosamente con detalles del premio
                    if (typeof appboyBridge !== 'undefined') {
                        const prizePoints = packData.total_prize_points;
                        const prizeName = packData.prize_sticker.sticker_name;
                        appboyBridge.logClick(`Pack Abierto - ${prizePoints} pts`);
                        appboyBridge.logCustomEvent('Album Pack Opened', {
                            album_id: packData.album_id,
                            prize_points: prizePoints,
                            prize_sticker: prizeName
                        });
                    }

                    state.tempRevealing = pickedStickers;
                    state.packData = packData; // Guardar datos completos para registro
                    state.view = 'summary';
                    state.isLoading = false;
                    renderApp();

                    // Registrar ganador en segundo plano
                    console.log('📝 Registrando ganador...');
                    setTimeout(() => {
                        registerWinner(packData);
                    }, 1000);
                }, 500);

            } catch (error) {
                console.error('❌ Error abriendo pack:', error);

                // Log de Braze: error al abrir pack
                if (typeof appboyBridge !== 'undefined') {
                    appboyBridge.logClick('Error al Abrir Pack');
                    appboyBridge.logCustomEvent('Album Pack Error', {
                        error_message: error.message,
                        album_id: ALBUM_ID
                    });
                }

                // Mostrar error al usuario
                alert('Error al abrir el pack. Por favor intenta de nuevo.\n\n' + error.message);

                state.isLoading = false;
                renderApp(); // Re-renderizar para quitar el loader

                // Resetear el pack container
                if (packContainer) {
                    packContainer.style.transform = '';
                    packContainer.style.opacity = '1';
                }
            }
        }

        function finalAction(e) {
            if (e) e.stopPropagation();
            if (typeof appboyBridge !== 'undefined') {
                appboyBridge.logClick('Obtener Más Packs');
            }
            const viewport = document.getElementById('app-viewport');
            const closeBtn = document.getElementById('global-close');
            closeBtn.style.display = 'none';
            viewport.innerHTML = `
                <div class="text-center text-white p-8">
                    <div class="mb-10 flex justify-center">
                        <div class="w-16 h-16 border-4 border-yellow-400 border-t-transparent rounded-full animate-spin loader"></div>
                    </div>
                    <p class="text-3xl font-black uppercase tracking-tighter leading-tight mb-4 text-center">Abriendo Desafíos BEES</p>
                    <p class="text-sm opacity-40 uppercase tracking-widest animate-pulse font-bold">Redirigiendo...</p>
                </div>
            `;
            setTimeout(() => {
                window.location.href = 'https://mybees.hn/rewards?storeId=a5bfd193-7ee7-48cb-96f7-5af632625ab2';
            }, 1500);
        }

        window.onload = renderApp;
