                const urlParams = new URLSearchParams(window.location.search);

                    return userId;

                // Si estamos en Braze/Appboy, intentar obtener el ID

                    try {

                        }
                    } catch (e) {
                        console.log('No se pudo obtener user_id de Braze');
                    }
                }

                // Fallback: generar ID temporal y guardarlo en sessionStorage
                let tempUserId = sessionStorage.getItem('temp_user_id');
                if (!tempUserId) {
                    tempUserId = 'temp_' + Date.now() + '_' + Math.random().toString(36).substr(2, 9);
                    sessionStorage.setItem('temp_user_id', tempUserId);
                }
                return tempUserId;

            // Llamar al API para abrir pack
            async function openPackFromAPI() {
                const userId = getUserId();

                try {
                    console.log('[API] Llamando a open-pack...');

                    const controller = new AbortController();
                    const timeoutId = setTimeout(() => controller.abort(), 10000);

                    const response = await fetch(`${API_BASE_URL}/open-pack`, {
                        method: 'POST',
                        headers: {
                            'Content-Type': 'application/json'
                        },
                        body: JSON.stringify({
                            album_id: ALBUM_ID,
                            user_id: userId
                        }),
                        signal: controller.signal
                    });

                    clearTimeout(timeoutId);
                    console.log('[API] Response status:', response.status);

                    if (!response.ok) {
                        const errorText = await response.text();
                        console.error('[API] Error response:', errorText);
                        throw new Error(`Error: ${response.status}`);
                    }

                    const data = await response.json();
                    console.log('[API] Data received:', data);

                    if (data.success) {
                        return data.data;
                    } else {
                        throw new Error(data.message || 'Error al abrir el pack');
                    }
                } catch (error) {
                    const isTimeout = error.name === 'AbortError';
                    console.error('[API] Error:', isTimeout ? 'TIMEOUT' : error.message);
                    throw error;
                }
            }

            // Registrar ganador en el API
            async function registerWinner(packData) {
                try {
                    console.log('[Registro] Enviando ganador:', packData);

                    const controller = new AbortController();
                    const timeoutId = setTimeout(() => controller.abort(), 10000);

                    const response = await fetch(`${API_BASE_URL}/register-winner`, {
                        method: 'POST',
                        headers: {
                            'Content-Type': 'application/json'
                        },
                        body: JSON.stringify(packData),
                        signal: controller.signal
                    });

                    clearTimeout(timeoutId);

                    if (!response.ok) {
                        const errorText = await response.text();
                        console.error('[Registro] Error:', errorText);
                        return false;
                    }

                    const data = await response.json();
                    console.log('[Registro] Ganador registrado:', data);
                    return true;
                } catch (error) {
                    const isTimeout = error.name === 'AbortError';
                    console.error('[Registro] Error:', isTimeout ? 'TIMEOUT' : error.message);
                    return false;
                }
            }

                            prize: sticker.prize_points,
                            brand: sticker.brand,
                            isPrize: sticker.is_prize,
                            prizeValue: sticker.prize_points
                        }));

                        if (typeof appboyBridge !== 'undefined') {
                            const prizePoints = packData.total_prize_points;

                            appboyBridge.logCustomEvent('Album Pack Opened', {
                                album_id: packData.album_id,
                                prize_points: prizePoints,
                                prize_sticker: prizeName
                            });
                        }

                        state.tempRevealing = pickedStickers;
                        state.packData = packData;
                        state.view = 'summary';
                        state.isLoading = false;
                        renderApp();

                        // Registrar ganador en segundo plano
                        console.log('📝 Registrando ganador...');
                        setTimeout(() => {
                            registerWinner(packData);
                        }, 1000);
                    }, 500);

                } catch (error) {
                    console.error('❌ Error abriendo pack:', error);

                    if (typeof appboyBridge !== 'undefined') {
                        appboyBridge.logClick('Error al Abrir Pack');
                        appboyBridge.logCustomEvent('Album Pack Error', {
                            error_message: error.message,
                            album_id: ALBUM_ID
                        });
                    }

                    // NO mostrar alert - manejo silencioso
                    state.isLoading = false;
                    renderApp();

                    if (packContainer) {
                        packContainer.style.transform = '';
                        packContainer.style.opacity = '1';
                    }
                }
            }

                if (typeof appboyBridge !== 'undefined') {
                    appboyBridge.logClick('Obtener Más Packs');

                const viewport = document.getElementById('app-viewport');
