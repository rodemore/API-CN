                modal.classList.add('hidden');
            }
        }

        function acceptTerms() {
            state.termsAccepted = true;
            toggleTermsModal(false);

            // Log de Braze: aceptar términos desde modal
            if (typeof appboyBridge !== 'undefined') {
                appboyBridge.logClick('Términos Aceptados - Botón Modal');

        function closeInApp() {
            if (typeof appboyBridge !== 'undefined') {
                appboyBridge.logClick('Cerrar Publicidad');
                appboyBridge.closeMessage();
                appboyBridge.logClick('Cerrar Inapp');
            }
        }

        // Llamar al API para abrir pack (usando fetch como milex.txt que funciona)
        async function openPackFromAPI() {
            const userId = getUserId();

            try {
                console.log('[API] Llamando a open-pack...');

                const controller = new AbortController();
                const timeoutId = setTimeout(() => controller.abort(), 10000);

                const response = await fetch(`${API_BASE_URL}/open-pack`, {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/json'
                    },
                    body: JSON.stringify({
                        album_id: ALBUM_ID,
                        user_id: userId
                    }),
                    signal: controller.signal
                });

                clearTimeout(timeoutId);
                console.log('[API] Response status:', response.status);

                if (!response.ok) {
                    const errorText = await response.text();
                    console.error('[API] Error response:', errorText);
                    throw new Error(`Error: ${response.status}`);
                }

                const data = await response.json();
                console.log('[API] Data received:', data);

                if (data.success) {
                    return data.data;
                } else {
                    throw new Error(data.message || 'Error al abrir el pack');
                }
            } catch (error) {
                const isTimeout = error.name === 'AbortError';
                console.error('[API] Error:', isTimeout ? 'TIMEOUT' : error.message);
                throw error;
            }
        }

        // Registrar ganador en el API (usando fetch como milex.txt que funciona)
        async function registerWinner(packData) {
            try {
                console.log('[Registro] Enviando ganador:', packData);

                const controller = new AbortController();
                const timeoutId = setTimeout(() => controller.abort(), 10000);

                const response = await fetch(`${API_BASE_URL}/register-winner`, {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/json'
                    },
                    body: JSON.stringify(packData),
                    signal: controller.signal
                });

                clearTimeout(timeoutId);

                if (!response.ok) {
                    const errorText = await response.text();
                    console.error('[Registro] Error:', errorText);
                    // No lanzamos error para no interrumpir la experiencia
                    return false;
                }

                const data = await response.json();
                console.log('[Registro] Ganador registrado:', data);
                return true;
            } catch (error) {
                const isTimeout = error.name === 'AbortError';
                console.error('[Registro] Error:', isTimeout ? 'TIMEOUT' : error.message);
                // No lanzamos error para no interrumpir la experiencia
                return false;
            }
        }

        async function startOpeningSequence() {
            if (!state.termsAccepted || state.isLoading) return;

            console.log('🎮 Iniciando apertura de pack...');

            if (typeof appboyBridge !== 'undefined') {
                appboyBridge.logClick('Abrir Pack');
            }

            // Activar estado de carga
            state.isLoading = true;
            renderApp(); // Re-renderizar para mostrar el loader

            const packContainer = document.getElementById('pack-container');

            try {
                // Llamar al API para abrir el pack
                console.log('📡 Llamando al API...');
                const packData = await openPackFromAPI();
                console.log('✅ Pack recibido del API:', packData);

                // Animación de cierre del pack
                if (packContainer) {
                    packContainer.style.transform = 'scale(3) rotate(20deg)';
                    packContainer.style.opacity = '0';
                }

                setTimeout(() => {
                    state.unopenedPacks--;

                    // Convertir los stickers del API al formato del frontend
                    const pickedStickers = packData.pack_stickers.map(sticker => ({
                        id: sticker.sticker_id,
                        url: sticker.sticker_url,
                        prize: sticker.prize_points,
                        brand: sticker.brand,
                        isPrize: sticker.is_prize,
                        prizeValue: sticker.prize_points
                    }));

                    console.log('🎨 Stickers procesados:', pickedStickers);

                    // Log de Braze: pack abierto exitosamente con detalles del premio
                    if (typeof appboyBridge !== 'undefined') {
                        const prizePoints = packData.total_prize_points;
                        const prizeName = packData.prize_sticker.sticker_name;
                        appboyBridge.logClick(`Pack Abierto - ${prizePoints} pts`);
                        appboyBridge.logCustomEvent('Album Pack Opened', {
                            album_id: packData.album_id,
                            prize_points: prizePoints,
                            prize_sticker: prizeName
                        });
                    }

                    state.tempRevealing = pickedStickers;
                    state.packData = packData; // Guardar datos completos para registro
                    state.view = 'summary';
                    state.isLoading = false;
                    renderApp();

                    // Registrar ganador en segundo plano
                    console.log('📝 Registrando ganador...');
                    setTimeout(() => {
                        registerWinner(packData);
                    }, 1000);
                }, 500);

            } catch (error) {
                console.error('❌ Error abriendo pack:', error);

                // Log de Braze: error al abrir pack
                if (typeof appboyBridge !== 'undefined') {
                    appboyBridge.logClick('Error al Abrir Pack');
                    appboyBridge.logCustomEvent('Album Pack Error', {
                        error_message: error.message,
                        album_id: ALBUM_ID
                    });
                }

                // NO mostrar alert - manejo silencioso del error
                // Simplemente resetear el estado para que el usuario pueda intentar de nuevo
                state.isLoading = false;
                renderApp(); // Re-renderizar para quitar el loader

                // Resetear el pack container
                if (packContainer) {
                    packContainer.style.transform = '';
                    packContainer.style.opacity = '1';
                }
            }
        }
